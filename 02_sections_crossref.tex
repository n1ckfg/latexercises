\title{Cross-References in \LaTeX}
\author{Author Name}
\date{\today}

\begin{document}

\maketitle

\begin{abstract}
This example demonstrates sectioning, labels, and cross-references
using \texttt{hyperref} and \texttt{cleveref} for smart references.
\end{abstract}

\section{Introduction}\label{sec:intro}
This is the introduction, labeled \texttt{sec:intro}.
We can reference it later with \cref{sec:intro} or \Cref{sec:intro}.

\section{Main Content}\label{sec:main}
Main content section.

\subsection{Background}\label{sec:background}
Background subsection.

\subsection{Methodology}\label{sec:methodology}
Methodology subsection with equation reference.

\begin{equation}\label{eq:sample}
E = mc^2
\end{equation}

Equation \cref{eq:sample} is Einstein's famous mass-energy equivalence.

\subsection{Results}\label{sec:results}
Results subsection with figure and table references.

\begin{figure}[htbp]
\centering
\fbox{\parbox{5cm}{\centering Sample Figure Content}}
\caption{A sample figure}\label{fig:sample}
\end{figure}

\begin{table}[htbp]
\centering
\begin{tabular}{lc}
\hline
Item & Value \\
\hline
A & 1 \\
B & 2 \\
\hline
\end{tabular}
\caption{A sample table}\label{tab:sample}
\end{table}

See \cref{fig:sample} and \cref{tab:sample}.

\section{Discussion}\label{sec:discussion}
Discussion section referencing multiple items:
\begin{itemize}
    \item Section: \cref{sec:background}
    \item Equation: \cref{eq:sample}
    \item Figure: \cref{fig:sample}
    \item Table: \cref{tab:sample}
    \item Multiple: \cref{sec:background,sec:methodology,sec:results}
\end{itemize}

\section{Conclusion}\label{sec:conclusion}
Conclusion referencing \cref{sec:intro}.

\appendix
\section{Appendix A}\label{app:a}
Appendix content.
